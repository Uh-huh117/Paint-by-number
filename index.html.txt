<script>
/**
 * Core Application Logic
 * Optimized for low lag utilizing Offscreen Canvas concepts and direct ImageData manipulation.
 */
const app = {
  projects: [],
  currentProject: null,
  
  // Canvas & Contexts
  canvas: null,
  ctx: null,
  offscreenCanvas: null,
  offscreenCtx: null,
  
  // Transform State (Pan/Zoom)
  transform: { x: 0, y: 0, scale: 1 },
  isDragging: false,
  isPainting: false,
  lastPointer: null,
  brushSize: 1,
  selectedColorIndex: 0,
  
  // Touch Gestures
  initialPinchDistance: null,
  initialScale: 1,

  init() {
    this.canvas = document.getElementById('main-canvas');
    this.ctx = this.canvas.getContext('2d', { alpha: false });
    this.offscreenCanvas = document.createElement('canvas');
    this.offscreenCtx = this.offscreenCanvas.getContext('2d', { willReadFrequently: true });
    
    this.loadProjects();
    this.renderGallery();
    this.setupEvents();
    
    // Resize handler
    window.addEventListener('resize', () => {
      if(this.currentProject) {
        this.resizeCanvas();
        this.requestRender();
      }
    });
  },

  showView(viewId) {
    document.querySelectorAll('.view').forEach(v => v.classList.remove('active'));
    document.getElementById(viewId).classList.add('active');
  },

  showLoading(text) {
    document.getElementById('loading-text').innerText = text;
    document.getElementById('loading').style.display = 'flex';
  },

  hideLoading() {
    document.getElementById('loading').style.display = 'none';
  },

  /* --- DATA MANAGEMENT --- */
  loadProjects() {
    const data = localStorage.getItem('pbn_projects');
    if (data) {
      try { this.projects = JSON.parse(data); } catch(e) { this.projects = []; }
    }
  },

  saveProjects() {
    try {
      localStorage.setItem('pbn_projects', JSON.stringify(this.projects));
    } catch(e) {
      alert("Storage limit reached! Please delete older projects to save new ones.");
    }
  },

  saveAndExit() {
    if (this.currentProject) {
      // Update thumbnail
      this.currentProject.thumbnail = this.canvas.toDataURL('image/jpeg', 0.5);
      const idx = this.projects.findIndex(p => p.id === this.currentProject.id);
      if(idx > -1) this.projects[idx] = this.currentProject;
      else this.projects.push(this.currentProject);
      
      this.saveProjects();
      this.currentProject = null;
    }
    this.renderGallery();
    this.showView('view-gallery');
  },

  /* --- GALLERY --- */
  renderGallery() {
    const container = document.getElementById('gallery-container');
    container.innerHTML = '';
    
    if (this.projects.length === 0) {
      container.innerHTML = '<p style="grid-column: 1/-1; text-align: center; color: #777;">No projects yet. Create one!</p>';
      return;
    }

    this.projects.forEach(p => {
      const total = p.width * p.height;
      let paintedCount = p.painted.split('').filter(c => c === '1').length;
      const progress = (paintedCount / total) * 100;
      
      const el = document.createElement('div');
      el.className = 'gallery-item';
      el.innerHTML = `
        <img src="${p.thumbnail}" alt="Thumbnail">
        <div style="display:flex; justify-content:space-between; align-items:center;">
          <strong>${p.width}x${p.height}</strong>
          <button onclick="event.stopPropagation(); app.deleteProject('${p.id}')" style="background:red; padding: 4px 8px; font-size: 12px;">X</button>
        </div>
        <div class="progress-bar"><div class="progress-fill" style="width: ${progress}%"></div></div>
      `;
      el.onclick = () => this.openProject(p.id);
      container.appendChild(el);
    });
  },

  deleteProject(id) {
    if(confirm("Delete this project?")) {
      this.projects = this.projects.filter(p => p.id !== id);
      this.saveProjects();
      this.renderGallery();
    }
  },

  /* --- PROJECT GENERATION (K-Means & Image Processing) --- */
  async generateProject() {
    const file = document.getElementById('image-upload').files[0];
    if (!file) return alert("Please select an image.");
    
    const maxColors = parseInt(document.getElementById('color-slider').value);
    const maxPixels = parseInt(document.getElementById('pixel-slider').value);
    
    this.showLoading("Processing image... (This may take a moment)");
    
    // Allow UI to update before heavy processing
    setTimeout(() => {
      const reader = new FileReader();
      reader.onload = (e) => {
        const img = new Image();
        img.onload = () => {
          this.processImage(img, maxColors, maxPixels);
        };
        img.src = e.target.result;
      };
      reader.readAsDataURL(file);
    }, 50);
  },

  processImage(img, maxColors, maxPixels) {
    // Calculate dimensions
    const ratio = img.width / img.height;
    let w = Math.round(Math.sqrt(maxPixels * ratio));
    let h = Math.round(maxPixels / w);
    
    const tempCanvas = document.createElement('canvas');
    tempCanvas.width = w; tempCanvas.height = h;
    const tCtx = tempCanvas.getContext('2d');
    tCtx.drawImage(img, 0, 0, w, h);
    
    const imgData = tCtx.getImageData(0, 0, w, h);
    const data = imgData.data;
    
    // Subsample for fast K-Means (max 2000 pixels)
    const samples = [];
    const step = Math.max(4, Math.floor(data.length / 4 / 2000) * 4);
    for(let i=0; i<data.length; i+=step) {
      if(data[i+3] > 128) samples.push([data[i], data[i+1], data[i+2]]); // ignore transparent
    }
    
    // Initialize Centroids
    let centroids = [];
    for(let i=0; i<maxColors; i++) {
      centroids.push(samples[Math.floor(Math.random() * samples.length)]);
    }
    
    // K-Means iterations (3 is enough for decent clustering quickly)
    for(let iter=0; iter<3; iter++) {
      const sums = Array.from({length: maxColors}, () => [0,0,0,0]);
      for(const p of samples) {
        let minD = Infinity, minIdx = 0;
        for(let c=0; c<maxColors; c++) {
          const d = Math.abs(p[0]-centroids[c][0]) + Math.abs(p[1]-centroids[c][1]) + Math.abs(p[2]-centroids[c][2]);
          if(d < minD) { minD = d; minIdx = c; }
        }
        sums[minIdx][0] += p[0]; sums[minIdx][1] += p[1]; sums[minIdx][2] += p[2]; sums[minIdx][3]++;
      }
      for(let c=0; c<maxColors; c++) {
        if(sums[c][3] > 0) {
          centroids[c] = [Math.round(sums[c][0]/sums[c][3]), Math.round(sums[c][1]/sums[c][3]), Math.round(sums[c][2]/sums[c][3])];
        }
      }
    }
    
    // Filter unused centroids and format palette
    centroids = centroids.filter(c => c !== undefined);
    
    // Map all pixels to closest centroid
    const pixelMap = new Array(w * h);
    for(let i=0; i<data.length; i+=4) {
      const p = [data[i], data[i+1], data[i+2]];
      let minD = Infinity, minIdx = 0;
      for(let c=0; c<centroids.length; c++) {
        const d = Math.abs(p[0]-centroids[c][0]) + Math.abs(p[1]-centroids[c][1]) + Math.abs(p[2]-centroids[c][2]);
        if(d < minD) { minD = d; minIdx = c; }
      }
      pixelMap[i/4] = minIdx;
    }

    const project = {
      id: 'proj_' + Date.now(),
      width: w,
      height: h,
      palette: centroids,
      pixels: pixelMap.join(','), // Store as comma separated string
      painted: new Array(w * h).fill(0).join(''), // String of 0s
      thumbnail: tempCanvas.toDataURL()
    };
    
    this.projects.push(project);
    this.saveProjects();
    this.hideLoading();
    this.openProject(project.id);
  },

  /* --- PAINT WORKSPACE --- */
  openProject(id) {
    this.currentProject = this.projects.find(p => p.id === id);
    if(!this.currentProject) return;
    
    this.offscreenCanvas.width = this.currentProject.width;
    this.offscreenCanvas.height = this.currentProject.height;
    
    this.showView('view-paint');
    this.resizeCanvas();
    
    // Reset transform to center
    const scale = Math.min(
      (this.canvas.width * 0.9) / this.currentProject.width, 
      (this.canvas.height * 0.9) / this.currentProject.height
    );
    this.transform = {
      x: (this.canvas.width - this.currentProject.width * scale) / 2,
      y: (this.canvas.height - this.currentProject.height * scale) / 2,
      scale: scale
    };
    
    // Find first unpainted color
    this.selectedColorIndex = 0;
    this.autoAdvanceColor();
    
    this.buildPaletteSidebar();
    this.updateProgress();
    this.requestRender();
  },

  resizeCanvas() {
    this.canvas.width = document.getElementById('paint-workspace').clientWidth;
    this.canvas.height = document.getElementById('paint-workspace').clientHeight;
  },

  buildPaletteSidebar() {
    const list = document.getElementById('palette-list');
    list.innerHTML = '';
    
    // Calculate unpainted counts
    const pixels = this.currentProject.pixels.split(',').map(Number);
    const painted = this.currentProject.painted.split('');
    const counts = new Array(this.currentProject.palette.length).fill(0);
    const totalCounts = new Array(this.currentProject.palette.length).fill(0);
    
    for(let i=0; i<pixels.length; i++) {
      totalCounts[pixels[i]]++;
      if(painted[i] === '0') counts[pixels[i]]++;
    }

    this.currentProject.palette.forEach((rgb, idx) => {
      // Skip colors that aren't in the image
      if (totalCounts[idx] === 0) return;

      const el = document.createElement('div');
      el.className = `color-item ${idx === this.selectedColorIndex ? 'active' : ''} ${counts[idx] === 0 ? 'completed' : ''}`;
      el.onclick = () => {
        this.selectedColorIndex = idx;
        this.buildPaletteSidebar();
        this.requestRender();
      };
      
      const hex = `#${rgb[0].toString(16).padStart(2,'0')}${rgb[1].toString(16).padStart(2,'0')}${rgb[2].toString(16).padStart(2,'0')}`;
      
      el.innerHTML = `
        <div class="color-swatch" style="background: ${hex}">${idx + 1}</div>
        <div class="color-info">Color ${idx + 1} <br><small>${totalCounts[idx] - counts[idx]} / ${totalCounts[idx]}</small></div>
      `;
      list.appendChild(el);
    });
  },

  updateBrushSize(val) {
    this.brushSize = parseInt(val);
    document.getElementById('brush-val').innerText = val;
    document.getElementById('brush-val2').innerText = val;
  },

  /* --- RENDERING ENGINE --- */
  // Extremely fast rendering using ImageData and hardware scaling
  requestRender() {
    if(!this.renderPending) {
      this.renderPending = true;
      requestAnimationFrame(() => this.render());
    }
  },

  render() {
    this.renderPending = false;
    
    const p = this.currentProject;
    const pixels = p.pixels.split(',').map(Number);
    const painted = p.painted.split('');
    const palette = p.palette;
    
    const highlightHex = document.getElementById('highlight-color').value;
    const hRGB = [
      parseInt(highlightHex.substring(1,3), 16),
      parseInt(highlightHex.substring(3,5), 16),
      parseInt(highlightHex.substring(5,7), 16)
    ];

    // Check completion mode
    let isComplete = painted.indexOf('0') === -1;
    const ws = document.getElementById('paint-workspace');
    if (isComplete) ws.classList.add('completed');
    else ws.classList.remove('completed');

    // 1. Generate Raw Image Data for the offscreen canvas
    const imgData = this.offscreenCtx.createImageData(p.width, p.height);
    const d = imgData.data;
    
    for (let i = 0; i < pixels.length; i++) {
      const idx = i * 4;
      const cIdx = pixels[i];
      
      if (painted[i] === '1' || isComplete) {
        // Painted pixel or Completed Mode (Show real color)
        d[idx] = palette[cIdx][0];
        d[idx+1] = palette[cIdx][1];
        d[idx+2] = palette[cIdx][2];
      } else if (cIdx === this.selectedColorIndex) {
        // Highlight active unpainted color
        d[idx] = hRGB[0];
        d[idx+1] = hRGB[1];
        d[idx+2] = hRGB[2];
      } else {
        // Unpainted background (White)
        d[idx] = 255;
        d[idx+1] = 255;
        d[idx+2] = 255;
      }
      d[idx+3] = 255; // Alpha
    }
    this.offscreenCtx.putImageData(imgData, 0, 0);

    // 2. Draw scaled to main canvas
    this.ctx.fillStyle = isComplete ? '#111' : '#e0e0e0';
    this.ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);
    
    this.ctx.imageSmoothingEnabled = false; // Keep pixels sharp
    this.ctx.save();
    this.ctx.translate(this.transform.x, this.transform.y);
    this.ctx.scale(this.transform.scale, this.transform.scale);
    
    this.ctx.drawImage(this.offscreenCanvas, 0, 0);
    this.ctx.restore();
  },

  /* --- INTERACTION (Paint, Pan, Zoom) --- */
  setupEvents() {
    const cvs = this.canvas;
    
    // Mouse Events
    cvs.addEventListener('mousedown', e => this.pointerDown(e, e.clientX, e.clientY));
    window.addEventListener('mousemove', e => this.pointerMove(e, e.clientX, e.clientY));
    window.addEventListener('mouseup', () => this.pointerUp());
    cvs.addEventListener('wheel', e => this.handleWheel(e), {passive: false});

    // Touch Events
    cvs.addEventListener('touchstart', e => {
      e.preventDefault();
      if(e.touches.length === 1) {
        this.pointerDown(e, e.touches[0].clientX, e.touches[0].clientY);
      } else if (e.touches.length === 2) {
        this.isPainting = false;
        this.isDragging = false;
        this.initialPinchDistance = Math.hypot(
          e.touches[0].clientX - e.touches[1].clientX,
          e.touches[0].clientY - e.touches[1].clientY
        );
        this.initialScale = this.transform.scale;
      }
    }, {passive: false});

    window.addEventListener('touchmove', e => {
      if(e.touches.length === 1) {
        this.pointerMove(e, e.touches[0].clientX, e.touches[0].clientY);
      } else if (e.touches.length === 2 && this.initialPinchDistance) {
        // Pinch Zoom
        e.preventDefault();
        const dist = Math.hypot(
          e.touches[0].clientX - e.touches[1].clientX,
          e.touches[0].clientY - e.touches[1].clientY
        );
        const centerX = (e.touches[0].clientX + e.touches[1].clientX) / 2;
        const centerY = (e.touches[0].clientY + e.touches[1].clientY) / 2;
        
        const scaleChange = dist / this.initialPinchDistance;
        this.applyZoom(this.initialScale * scaleChange, centerX, centerY);
      }
    }, {passive: false});

    window.addEventListener('touchend', () => {
      this.pointerUp();
      this.initialPinchDistance = null;
    });
  },

  pointerDown(e, x, y) {
    // If middle click or shift key, pan
    if(e.button === 1 || e.shiftKey) {
      this.isDragging = true;
      this.lastPointer = {x, y};
    } else {
      // Paint
      this.isPainting = true;
      this.paintAt(x, y);
    }
  },

  pointerMove(e, x, y) {
    if(this.isDragging) {
      this.transform.x += (x - this.lastPointer.x);
      this.transform.y += (y - this.lastPointer.y);
      this.lastPointer = {x, y};
      this.requestRender();
    } else if (this.isPainting) {
      this.paintAt(x, y);
    }
  },

  pointerUp() {
    if(this.isPainting) {
      this.updateProgress();
      this.autoAdvanceColor();
    }
    this.isPainting = false;
    this.isDragging = false;
  },

  handleWheel(e) {
    e.preventDefault();
    const zoomFactor = 1.1;
    const scale = e.deltaY < 0 ? this.transform.scale * zoomFactor : this.transform.scale / zoomFactor;
    this.applyZoom(scale, e.clientX, e.clientY);
  },

  applyZoom(newScale, pointerX, pointerY) {
    // Math to zoom into pointer location
    newScale = Math.max(0.5, Math.min(newScale, 50)); // clamp zoom
    const cvsRect = this.canvas.getBoundingClientRect();
    const mouseX = pointerX - cvsRect.left;
    const mouseY = pointerY - cvsRect.top;
    
    this.transform.x = mouseX - (mouseX - this.transform.x) * (newScale / this.transform.scale);
    this.transform.y = mouseY - (mouseY - this.transform.y) * (newScale / this.transform.scale);
    this.transform.scale = newScale;
    
    this.requestRender();
  },

  paintAt(screenX, screenY) {
    const rect = this.canvas.getBoundingClientRect();
    const x = (screenX - rect.left - this.transform.x) / this.transform.scale;
    const y = (screenY - rect.top - this.transform.y) / this.transform.scale;
    
    const pX = Math.floor(x);
    const pY = Math.floor(y);
    
    if (pX < 0 || pX >= this.currentProject.width || pY < 0 || pY >= this.currentProject.height) return;

    let paintedArr = this.currentProject.painted.split('');
    const pixels = this.currentProject.pixels.split(',').map(Number);
    let changed = false;

    // Apply brush size
    const offset = Math.floor(this.brushSize / 2);
    for(let dy = -offset; dy <= offset; dy++) {
      for(let dx = -offset; dx <= offset; dx++) {
        const targetX = pX + dx;
        const targetY = pY + dy;
        
        if (targetX >= 0 && targetX < this.currentProject.width && targetY >= 0 && targetY < this.currentProject.height) {
          const i = targetY * this.currentProject.width + targetX;
          // Only paint if it matches selected color AND isn't already painted
          if (pixels[i] === this.selectedColorIndex && paintedArr[i] === '0') {
            paintedArr[i] = '1';
            changed = true;
          }
        }
      }
    }

    if(changed) {
      this.currentProject.painted = paintedArr.join('');
      this.requestRender();
    }
  },

  updateProgress() {
    const total = this.currentProject.width * this.currentProject.height;
    let paintedCount = this.currentProject.painted.split('').filter(c => c === '1').length;
    const percent = Math.floor((paintedCount / total) * 100);
    document.getElementById('progress-text').innerText = `${percent}%`;
    this.buildPaletteSidebar();
  },

  autoAdvanceColor() {
    const pixels = this.currentProject.pixels.split(',').map(Number);
    const painted = this.currentProject.painted.split('');
    
    // Check if current is done
    let currentDone = true;
    for(let i=0; i<pixels.length; i++) {
      if(pixels[i] === this.selectedColorIndex && painted[i] === '0') {
        currentDone = false; break;
      }
    }
    
    if(currentDone) {
      // Find next incomplete color
      const counts = new Array(this.currentProject.palette.length).fill(0);
      for(let i=0; i<pixels.length; i++) {
        if(painted[i] === '0') counts[pixels[i]]++;
      }
      
      const nextIdx = counts.findIndex(c => c > 0);
      if(nextIdx !== -1 && nextIdx !== this.selectedColorIndex) {
        this.selectedColorIndex = nextIdx;
        this.buildPaletteSidebar();
        this.requestRender();
      }
    }
  }
};

window.onload = () => app.init();
